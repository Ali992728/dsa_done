void islandsAndTreasure(vector<vector<int>>& grid) {
        int n=grid.size();
        int m=grid[0].size();
    
        queue<pair<pair<int,int>,int>> q;
        vector<vector<int>> vis(n,vector<int>(m,0));

        for(int i=0;i<n;i++) {
            for(int j=0;j<m;j++) {
                if(grid[i][j]==0) {
                    q.push({{i,j},0});
                    vis[i][j]=1;
                }
            }
        }

        while(!q.empty()) {
            int i=q.front().first.first;
            int j=q.front().first.second;
            int cnt=q.front().second;
            q.pop();
            grid[i][j]=cnt;

            if(i+1<n && grid[i+1][j]==INT_MAX && !vis[i+1][j]) {
                q.push({{i+1,j},cnt+1});
                vis[i+1][j]=1;
            }
            if(i-1>=0 && grid[i-1][j]==INT_MAX && !vis[i-1][j]) {
                q.push({{i-1,j},cnt+1});
                vis[i-1][j]=1;
            }
            if(j+1<m && grid[i][j+1]==INT_MAX && !vis[i][j+1]) {
                q.push({{i,j+1},cnt+1});
                vis[i][j+1]=1;
            }
            if(j-1>=0 && grid[i][j-1]==INT_MAX && !vis[i][j-1]) {
                q.push({{i,j-1},cnt+1});
                vis[i][j-1]=1;
            }
        }
    }
