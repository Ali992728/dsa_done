private:
    void dfs(int src,vector<bool> &vis,vector<vector<int>>& adj,int cnt) {
        vis[src]=true;

        for(auto v:adj[src]) {
            if(!vis[v]) {
                dfs(v,vis,adj,cnt);
            }
        }
    }
public:
    int countComponents(int n, vector<vector<int>>& edges) {
        vector<vector<int>> adj(n);
        for(int i=0;i<edges.size();i++) {
            int v=edges[i][0];
            int u=edges[i][1];

            adj[u].push_back(v);
            adj[v].push_back(u);
        }

        vector<bool> vis(n,false);

        int cnt=0;

        for(int i=0;i<n;i++) {
            if(!vis[i]) {
                dfs(i,vis,adj,cnt);
                cnt++;
            }
        }

        return cnt;
    }
